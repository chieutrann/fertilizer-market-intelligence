MoM % = 
VAR CurrentDate =
    MAX(commodity[Date])

VAR CurrentPrice =
    CALCULATE(
        MAX(commodity[Price]),
        commodity[Date] = CurrentDate
    )

VAR PreviousDate =
    MAXX(
        FILTER(
            ALL(commodity[Date]),
            commodity[Date] < CurrentDate
        ),
        commodity[Date]
    )

VAR PreviousPrice =
    CALCULATE(
        MAX(commodity[Price]),
        commodity[Date] = PreviousDate
    )

RETURN
    round(DIVIDE(CurrentPrice - PreviousPrice, PreviousPrice)*100,2)  & " %"
